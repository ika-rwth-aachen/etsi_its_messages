55inline uint64_t getUnixSecondsOfYear(const uint64_t unixSecond) {
56 time_t ts = static_cast<time_t>(unixSecond);
57 struct tm* timeinfo = gmtime(&ts);
58
59 timeinfo->tm_sec = 0;
60 timeinfo->tm_min = 0;
61 timeinfo->tm_hour = 0;
62 timeinfo->tm_mday = 1;
63 timeinfo->tm_mon = 0;
64
65 return timegm(timeinfo);
66}

75inline uint64_t getUnixNanosecondsFromMinuteOfTheYear(const MinuteOfTheYear& moy,
76 const uint64_t unix_nanoseconds_estimate) {
77 const uint64_t unix_seconds_of_year =
78 getUnixSecondsOfYear(static_cast<uint64_t>(unix_nanoseconds_estimate * 1e-9));
79 return (unix_seconds_of_year + static_cast<uint64_t>(moy.value) * 60ULL) * 1000000000ULL;
80}

89inline uint64_t getUnixNanosecondsFromRTCMcorrections(const RTCMcorrections& rtcm,
90 const uint64_t unix_nanoseconds_estimate) {
91 return getUnixNanosecondsFromMinuteOfTheYear(getMinuteOfTheYear(rtcm), unix_nanoseconds_estimate);

101inline uint64_t getUnixNanoseconds(const RTCMEM& rtcmem, const uint64_t unix_nanoseconds_estimate) {
102 return getUnixNanosecondsFromRTCMcorrections(rtcmem.rtcmc, unix_nanoseconds_estimate);

114inline double interpretLatitudeInDegrees(const Latitude& lat) {
115 if (lat.value == Latitude::UNAVAILABLE) {
116 return std::numeric_limits<double>::quiet_NaN();
117 }
118 return static_cast<double>(lat.value) * 1e-7;
119}
