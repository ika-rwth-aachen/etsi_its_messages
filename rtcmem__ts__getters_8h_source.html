551 inline bool getGNSSstatusBit(const GNSSstatus& status, const uint8_t bit_index) {
552 throwIfOutOfRange(bit_index, static_cast<uint8_t>(0), static_cast<uint8_t>(GNSSstatus::SIZE_BITS - 1), "GNSSstatus bit index");
553
554 if (status.value.empty()) {
555 return false;
556 }
557
558 const uint8_t mask = static_cast<uint8_t>(1u << bit_index);
559 return (status.value[0] & mask) != 0;
560 }
