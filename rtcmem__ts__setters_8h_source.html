552 inline void setGNSSstatusBit(GNSSstatus& status, const uint8_t bit_index, const bool value) {
553 throwIfOutOfRange(bit_index, static_cast<uint8_t>(0), static_cast<uint8_t>(GNSSstatus::SIZE_BITS - 1), "GNSSstatus bit index");
554
555 ensureGNSSstatusStorage(status);
556 const uint8_t mask = static_cast<uint8_t>(1u << bit_index);
557
558 if (value) {
559 status.value[0] = static_cast<uint8_t>(status.value[0] | mask);
560 } else {
561 status.value[0] = static_cast<uint8_t>(status.value[0] & ~mask);
562 }
563 }
